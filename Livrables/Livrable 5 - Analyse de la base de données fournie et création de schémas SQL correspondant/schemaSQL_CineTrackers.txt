//Chaque premier attribut est une clé primaire

PERSONNES (id_personne, nom, prenom, naissance, deces)

PROFESSION_CATEGORIES (id_profession, nom_profession)

JOB (id_job, nom_job)

GENRE (id_genre, nom_genre)

TYPE_OEUVRE (id_type, nom_type)

REGION (id_region, nom_region)

LANGUE (id_langue, nom_langue)

OEUVRE (id_titre, titre_original, #id_type, date_sortie, date_fin, est_adulte, duree)
la clé étrangère "id_type" de la table OEUVRE correspond à la clé primaire de la table TYPE_OEUVRE

OEUVRE_GENRE (#id_titre, #id_genre) --> couple de clé primaire
les clés étrangères respectivement "id_titre" et "id_genre" de la table OEUVRE_GENRE correspondent respectivement aux clés primaires des tables OEUVRE et GENRE

EPISODE (#id_episode, #id_serie, numero_saison, numero_episode)
les clés étrangères "id_serie" et "id_episode" de la table EPISODE correspondent à la clé primaire de la table OEUVRE

PARTICIPATION_OEUVRE (id_participation, ordre, #id_titre, #id_personne, #id_profession, #id_job, personnage_joue)
les clés étrangères respectivement "id_titre", "id_personne", "id_profession" et "id_job" de la table PARTICIPATION_OEUVRE correspondent respectivement aux clés primaires des tables OEUVRE, PERSONNES, PROFESSION_CATEGORIES et JOB

EVALUATION (#id_titre, moyenne_evaluation, nb_votes)
la clé étrangère "id_titre" de la table EVALUATION correspond à la clé primaire de la table OEUVRE

ADAPTATION_INTERNATIONAL (id_adaptation, #id_titre, titre_adapte, #id_region, #id_langue)
les clés étrangères respectivement "id_titre", "id_region" et "id_langue" de la table ADAPTATION_INTERNATIONAL correspondent respectivement aux clés primaires des tables OEUVRE, REGION et LANGUE
